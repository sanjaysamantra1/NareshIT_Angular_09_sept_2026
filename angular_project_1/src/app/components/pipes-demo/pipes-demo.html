Let's Learn Pipes

uppercase, lowercase, titlecase pipe::

customerName: {{customerName}}
customerName: {{customerName | uppercase}}
customerName: {{customerName | lowercase}}
customerName: {{customerName | titlecase}}

currency pipe::

salary: {{salary}}
salary: {{salary | currency}}
salary: {{salary | currency : 'INR'}}
salary: {{salary | currency : '€'}}
salary: {{salary | currency : '¥'}}

Date pipe::

{{dateObj}}
{{dateObj | date}}
{{dateObj | date : 'short'}}
{{dateObj | date : 'medium'}}
{{dateObj | date : 'long'}}
{{dateObj | date : 'full'}}
==================
{{ dateObj | date : 'shortDate' }}
{{ dateObj | date : 'mediumDate' }}
{{ dateObj | date : 'longDate' }}
{{ dateObj | date : 'fullDate' }}
=================
{{ dateObj | date : 'shortTime' }}
{{ dateObj | date : 'mediumTime' }}
{{ dateObj | date : 'longTime' }}
{{ dateObj | date : 'fullTime' }}
==============
{{ dateObj | date : 'dd-MMM-yyyy' }}
{{ dateObj | date : 'hh:mm:ss a' }}
{{ dateObj | date: "fullTime":"UTC" }}
{{ dateObj | date: "fullTime":"IST" }}
{{ dateObj | date: "EEEE" }}

Number & percent pipe::

{{100000000 | number}}
{{ 12.111222333 | number: "3.2-5" }}
{{ 12.1 | number: "3.2-5" }}
{{ 0.5 | number:'3.2-5'}}
=================
{{ 2.5 | percent}}
{{ 2.5 | percent:'2.2-5'}}
{{ 0.024 | percent}}
{{ 0.026 | percent}}
{{ 0.25 | percent}}

Json pipe::

{{user}}
{{user | json}}

keyvalue pipe::

@for (entry of user | keyvalue; track $index) {
{{entry.key}} ========= {{entry.value}}
}

slice pipe::

{{cars | slice : 2 }}
{{cars | slice : 2 : 4}}
{{'Angular is a Framework' | slice : 0 : 7}}
{{'Angular is a Framework' | slice : 0 : -4}}

Pipe Chaining::

{{'Angular is a Framework' | slice : 0 : 7 | uppercase }}
{{'Angular is a Framework' | slice : 0 : 7 | lowercase }}

Custom Pipe(ordinal):

number:{{num}}, ordinal value: {{num | ordinal}}
